const PB = require('../../config/combat-presentation.js');

## 二、按塔色显示伤害数字（中影响）

### 接线点（3 处，全部在 owner 手里）

1. **每局创建实例**（main.js 战斗初始化）：
```js
const DamageNumbers = require('../game/fx/damage-numbers.js');
this.dmgNumbers = DamageNumbers.create();     // 退出战斗 / 重开时 this.dmgNumbers.clear()
```
2. **发事件**——唯一伤害入口 `combat/damage.js`，扣血后、死亡递归回调前（现 damage.js:84 `e.hp -= v;` 处）：
```js
const hpBefore = e.hp;          // ← 需在 -= 之前取（显示实际扣除的存活血量）
e.hp -= v;
// …
game.dmgNumbers.push({
  enemyId: e.id, x: e.x, y: e.y - 40,
  sourceKey: srcKey,            // ← 用本次伤害的来源塔，不用 e.lastDmgKey 反推
  amount: Math.min(v, hpBefore),// 护甲/易伤后的实际扣除，限制过量伤害
  kind: isDotTick ? 'dot' : 'hit',   // 灼烧/中毒/流血跳数='dot'（可合并），直伤/技能='hit'
});
```
   腐化直接真伤（`combat/statuses.js:166`）同样补一条 `kind:'true'` 事件——**不漏、不重复**。
3. **每帧与绘制**：
```js
// 战斗 update（暂停/三选一/结算不推进）：
this.dmgNumbers.step(dt);
// 渲染：敌人层之后
this.dmgNumbers.drawTo(ctx, { text: ui.text });
```

### 行为要点（模块已实现，测试覆盖）

- 配色按 `sourceKey` 六塔映射（任务单指定色值），未知来源中性色 `#cfd8e3`，**不猜塔**；
- 显示实际扣除量（护甲/易伤/真火/标记加成后），≥1万 转「1.2万」，≥100 万取整，超大兜底「9999万+」，0/负数显示 1；
- 密集 DoT 合并：同敌人 + 同来源 + 0.12s 窗口内 → 累计为一条（`stats.merged` 计数）；直伤永不合并；
- 同一敌人同时多条自动纵向错位（22px/条，最多 4 条）；
- 表现上限 60 条，超出丢最旧并计数——**只影响表现，真实伤害/击杀/奖励不变**；
- 横向漂移用确定性哈希，不消耗 `Math.random`（不破坏 RNG 顺序类测试）。

## 三、连消全格弹道（替换旧连消攻击）

### 拍板规则的落点

| 规则 | 模块行为 |
|---|---|
| 新弹道**替换**旧连消攻击 | Run 侧 `_chainBonus` 的 `spawnChainShots` 调用点删除，改由本模块输出驱动（见下） |
| 保留**经验与充能** | `applyMatch` / `_blastReward` / 充能结算**一行不改**；模块不产出经验与充能 |
| 未出战颜色塔 → 充能回退 | 旧行为保留：模块输出 `kind:'charge'` 条目，由 Run 侧执行原 `_chargeAmmo` 逻辑 |
| 彩虹无色格 | 默认 `skip`（绝不默认红塔）；需要表现时 `policy.rainbow='neutral'` 出通用白弹 |
| 同步同格去重 / 跨步可重发 | step+row+col 去重；后续步骤重新出现即再发 |
| 第 4 组阈值不变 | 默认 `startGroup=4`（连锁风暴全局技时 Run 侧传 `startGroup:2`） |
| 视觉预算 | 单步 >120 条打 `overflow` 标记并汇总 `overflowBatches`，**不丢条目**（伤害不能静默丢） |

### 接线点（Run / 动画 / Combat 三处）

1. **Run.trySwap**（run.js:466 stepList 循环内）：
```js
const ChainAttacks = require('../skills/chain-attacks.js');
// 循环前：
const chainPlan = ChainAttacks.plan({
  steps: ret.resolved.stepList,
  startGroup: chainstormActive ? 2 : undefined,
  towerOfColor: (c) => CONFIG.COLORS[c] && CONFIG.COLORS[c].tower,
  activeTowers: this.slotOrder,
  layout: { originY: this.boardY },
});
// 原有经验/充能结算不动；_chainBonus 的 attack/skill 分支删除：
//   - 'attack'（spawnChainShots）→ 由 chainPlan 接管
//   - 'skill'（castPeriodicNow 随机周期技）→ 一并下架（不再叠加旧奖励）；
//     若要保留，须 owner 另拍板，不得与新弹道同格叠加
// kind==='charge' 的条目在此处直接执行旧充能回退（原 _chargeAmmo 逻辑）
chainPlan.shots.filter(s => s.stepIndex === si && s.kind === 'charge')
  .forEach(() => this._chargeAmmo('shuffle', half), /* 原口径 */);
```
2. **逐级动画发射**（match3.js 连锁动画阶段，**不要**在 trySwap 同帧全发）：
```js
// 每级动画开始时：
(plan.shotsByStep[si] || []).filter(s => s.kind === 'tower')
  .forEach(s => combat.spawnChainShotFromCell(s));   // 复用现 spawnChainShots 单发路径
```
3. **Combat**：伤害只在**命中时**结算一次（消除阶段不扣、命中不重复）；发奖走原死亡钩子；退出战斗/重开清空在飞弹道队列；暂停/三选一期间弹道冻结不积压重放。

### 建议第一批只做「塔色弹道 + 充能回退」，彩虹通用弹与溢出聚合第二批接（模块已支持，接线可分期）。
